\chapter{曲線の基本定理}

\section{フレネ・セレの公式}

小節\ref{subsection:differential-of-frames}で示した公式を並べてみましょう。
\begin{align*}
  \frac{de_1(s)}{ds}&=\kappa(s)e_2(s)\\
  \frac{de_2(s)}{ds}&=-\kappa(s)e_1(s)+\tau(s)e_3(s)\\
  \frac{de_3(s)}{ds}&=-\tau(s)e_2(s)
\end{align*}
なんだか行列みたいですね。
行列みたいに書いてみましょう。
\[
  \frac{d}{ds}
  \begin{pmatrix}
    e_1(s)\\
    e_2(s)\\
    e_3(s)
  \end{pmatrix}
  =
  \begin{pmatrix}
    0 & \kappa(s) & 0 \\
    -\kappa(s) & 0 & \tau(s) \\
    0 & -\tau(s) & 0
  \end{pmatrix}
  \begin{pmatrix}
    e_1(s)\\
    e_2(s)\\
    e_3(s)
  \end{pmatrix}
\]
これは小節\ref{subsection:differential-of-frames}の計算結果を行列の形に並べたに過ぎないですが、曲線を文字通り特徴づける、意味ある等式です。
とても偉いので、名前を付けましょう。
\begin{definition}
  $I$を区間、$\gamma:I\to\mathbb{R}^3$を弧長パラメータ付け可能な$C^3$級曲線であって、弧長パラメータ$s=s(t)$に関して$d^2\gamma(t(s))/ds^2\neq0$を満たすとする。
  このとき直交行列
  \[
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
  \]
  を\textbf{フレネ標構}と呼ぶ。
\end{definition}
各$e_i(s)$は$\mathbb{R}^3$の元なので、フレネ標構は3次行列です。
また
\[
  e_i(s)\cdot e_j(s)=\begin{cases}
    1 & (i=j)\\
    0 & (i\neq j)
  \end{cases}
\]
なので、フレネ標構は直交行列なわけです。

フレネ標構に関して、次の定理が成り立つことは既に証明しました。
これは\textbf{フレネ・セレの公式}と呼ばれます。
\begin{theorem}[フレネ・セレの公式]
  $I$を区間、$\gamma:I\to\mathbb{R}^3$を弧長パラメータ付け可能な$C^3$級曲線であって、弧長パラメータ$s=s(t)$に関して常に$d^2\gamma(t(s))/ds^2\neq0$を満たすとする。
  また、
  \begin{align*}
    e_1(s)&:=\frac{d\gamma(t(s))}{ds}\\
    e_2(s)&:=\frac{d^2\gamma(t(s))/ds^2}{|d^2\gamma(t(s))/ds^2|}\\
    e_3(s)&:=e_1(s)\times e_2(s)
  \end{align*}
  をそれぞれ$\gamma$の単位接ベクトル、主法線ベクトル、従法線ベクトルとし、
  \begin{align*}
    \kappa(s)&:=\left|\frac{d^2\gamma(t(s))}{ds^2}\right|\\
    \tau(s)&:=e_3(s)\cdot\frac{de_2(s)}{ds}
  \end{align*}
  をそれぞれ$\gamma$の曲率、捩率とする。
  このとき、フレネ標構に関して
  \[
    \frac{d}{ds}
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
    =
    \begin{pmatrix}
      0 & \kappa(s) & 0 \\
      -\kappa(s) & 0 & \tau(s) \\
      0 & -\tau(s) & 0
    \end{pmatrix}
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
  \]
  が成り立つ。
\end{theorem}

\section{曲線の基本定理}

幾何学の大目標は、\textbf{図形をある基準で分類したい}というものです。
今回は中学数学以来慣れ親しんでいるであろう「合同」を基準にしてみます。
何を隠そう曲率と捩率は、\textbf{曲線の合同変換のクラスを完全に分類する}のです。
そのような意味で、曲率と捩率は曲線論において最も重要な\textbf{不変量}なのです。

その前に合同という分類がいかなるものか、大学数学流にしっかりと定義しましょう。
\begin{definition}
  $(\gamma,I)$と$(\gamma',I')$を弧長パラメータづけ可能な曲線とし、$s\in s(I)$および$s'\in s'(I')$をその弧長パラメータとする。
  $(\gamma,I)$と$(\gamma',I')$が互いに\textbf{合同}であるとは、次を満たす時をいう。
  \begin{itemize}
    \item $\sigma\in\{\pm1\}$と$s_0\in s(I)$が存在して、写像$s(I)\to s'(I');s\mapsto \sigma s+s_0$は全単射
    \item ある3次直交行列$M$と$p\in\mathbb{R}^3$が存在して、
    \[
      \gamma'(t(\sigma s+s_0)) = M(\gamma(t(s))) + p
    \]
  \end{itemize}
\end{definition}

まず示したい定理の一つは、\textbf{2つの曲線が合同なら、それらの曲率と捩率は一致する}です。
これは\textbf{曲線の基本定理}と呼ばれるものの一部になります。
まずはPart1として次を示しましょう。
\begin{theorem}[曲線の基本定理(Part1)]
  曲線$(\gamma,I)$と$(\gamma',I')$に関して以下の仮定を満たすとする。
  \begin{itemize}
    \item ともに弧長パラメータづけ可能な$C^3$級曲線
    \item それぞれの弧長パラメータ$s\in s(I),s'\in s'(I')$について、常に$\frac{d^2\gamma(t(s))}{ds^2}\neq0$、$\frac{d^2\gamma'(t'(s'))}{ds'^2}\neq0$
    \item $\kappa(s)$、$\tau(s)$は$(\gamma,I)$の曲率と捩率
    \item $\kappa'(s')$、$\tau'(s')$は$(\gamma',I')$の曲率と捩率
  \end{itemize}
  さらに二つの曲線は互いに合同であるとする。すなわち、
  \begin{itemize}
    \item ある開区間$J\subset s(I)$と開区間$J'\subset s'(I')$、および、ある$s_0\in\mathbb{R}と\sigma\in\{\pm1\}$が存在し、\\
    写像 $J\to J';s\mapsto\sigma s+s_0$は全単射
    \item ある3次直交行列$M$と$p\in\mathbb{R}^3$が存在して、任意の$s\in J$に対して
    \[
      \gamma'(t'(\sigma s+s_0)) = M(\gamma(t(s))) + p
    \]
  \end{itemize}
  このとき、任意の$s\in J$に対して
  \[
    \kappa'(\sigma s+s_0)=\kappa(s),\quad\tau'(\sigma s+s_0)=\det(M)\tau(s)
  \]
  を満たす。
\end{theorem}
\begin{proof}
  曲率は$\gamma$の二階微分の長さであったから、
  与式
  \[
    \gamma'(t'(\sigma s+s_0)) = M(\gamma(t(s))) + p
  \]
  において、両辺の二階微分を行っても、それらは一致することからアプローチする。

  まず左辺の二階微分は、$u=\sigma s+s_0$の変数変換で
  \begin{align*}
    \frac{d\gamma'(t'(u))}{ds}&=\frac{du}{ds}\frac{d\gamma'(t'(u))}{du}=\sigma\frac{d\gamma'(t'(u))}{du}\\
    \frac{d^2\gamma'(t'(u))}{ds^2}&=\frac{du}{ds}\frac{d}{du}\frac{d\gamma'(t'(u))}{du}=\sigma^2\frac{d^2\gamma'(t'(u))}{du^2}=\frac{d^2\gamma'(t'(u))}{du^2}
  \end{align*}
  と計算できる。
  ここで$\sigma\in\{\pm1\}$なので、必ず$\sigma^2=1$であることを使用した。
  ゆえに
  \[
    \frac{d^2\text{(LHS)}}{ds^2}=\left|\frac{d^2\gamma'(t'(u))}{du^2}\right|=\kappa'(u)
  \]

  一方、与式の右辺を二階微分すると、$M$はただの定値の直交行列ゆえ
  \begin{align*}
    \frac{d^2(M\gamma(t(u))+p)}{ds^2}&=M\frac{d^2\gamma(t(s))}{ds^2}
  \end{align*}
  となり、
  \[
    \left|\frac{d^2\text{(RHS)}}{ds^2}\right|=\left|M\frac{d^2\gamma(t(s))}{ds^2}\right|
  \]
  となる。
  ところが直交行列をかけることは、ベクトルの長さを変えないので、結局は
  \[
    \left|\frac{d^2\text{(RHS)}}{ds^2}\right|=\kappa(s)
  \]
  以上より、
  \[
    \kappa(s)=\kappa'(u)=\kappa'(\sigma s+s_0)
  \]
  が従う。

  捩率に関しては、定義式
  \[
    \tau(s):=\frac{\left(\frac{d\gamma(t(s))}{ds}\times\frac{d^2\gamma(t(s))}{ds^2}\right)\cdot\frac{d^3\gamma(t(s))}{ds^3}}{\left|\frac{d^2\gamma(t(s))}{ds^2}\right|^2}
  \]
  を用いる。
  すぐにわかることは与式
  \[
    \gamma'(t'(\sigma s+s_0)) = M(\gamma(t(s))) + p
  \]
  の右辺は、微分しても$M$が掛かるだけである。
  また、ベクトル三重積の公式から、捩率の定義式の分子は、
  \[
    \frac{d\gamma(t(s))}{ds},\frac{d^2\gamma(t(s))}{ds^2},\frac{d^3\gamma(t(s))}{ds^3}
  \]
  を並べた行列の行列式である。
  従って与式の右辺に対して捩率の定義式をあてはめると
  \[
    \frac{\left(M\frac{d\gamma(t(s))}{ds}\times M\frac{d^2\gamma(t(s))}{ds^2}\right)\cdot M\frac{d^3\gamma(t(s))}{ds^3}}{\left|M\frac{d^2\gamma(t(s))}{ds^2}\right|^2}=\det(M)\frac{\left(\frac{d\gamma(t(s))}{ds}\times\frac{d^2\gamma(t(s))}{ds^2}\right)\cdot\frac{d^3\gamma(t(s))}{ds^3}}{\left|\frac{d^2\gamma(t(s))}{ds^2}\right|^2}=\det(M)\tau(s)
  \]
  となる。
  また与式の左辺の微分に対しても、$u=\sigma s+s_0$の変数変換によって、微分する変数は変わるけれども、$\sigma$が微分回数分かかるだけである。
  従って、これも捩率の定義式にあてはめると
  \[
    \frac{\left(\frac{d\gamma'(t'(u))}{ds}\times\frac{d^2\gamma'(t'(u))}{ds^2}\right)\cdot\frac{d^3\gamma'(t'(u))}{ds^3}}{\left|\frac{d^2\gamma'(t'(u))}{ds^2}\right|^2}=\frac{\left(\sigma\frac{d\gamma'(t'(u))}{du}\times\sigma^2\frac{d^2\gamma'(t'(u))}{du^2}\right)\cdot\sigma^3\frac{d^3\gamma'(t'(u))}{du^3}}{\left|\sigma^2\frac{d^2\gamma'(t'(u))}{du^2}\right|^2}
  \]
  $\sigma\in\{\pm1\}$の偶数乗は必ず1になることに気を付ければ、結局
  \[
    \frac{\left(\frac{d\gamma'(t'(u))}{ds}\times\frac{d^2\gamma'(t'(u))}{ds^2}\right)\cdot\frac{d^3\gamma'(t'(u))}{ds^3}}{\left|\frac{d^2\gamma'(t'(u))}{ds^2}\right|^2}=\tau'(u)
  \]
  となります。
  これらの捩率の定義式から得られた関数は、定理の与式から等しくなるべきであるから、結局
  \[
    \det(M)\tau(s)=\tau'(u)=\tau'(\sigma s+s_0)
  \]
  が確かめられた。
\end{proof}

合同変換で写りあう曲線が二つあたえられたとき、それらの曲率と捩率がそれぞれ等しいことが確かめられました。
数学者の性として次に気になることは、その逆、つまり、\textbf{曲率と捩率がそれぞれ等しい曲線が二つ与えられたとき、それらは合同変換で写りあうか？}という問題です。
これは肯定的に解くことができます。
\begin{theorem}[曲線の基本定理(Part2)]
  曲線$(\gamma,I)$と$(\gamma',I')$に関して以下の仮定を満たすとする。
  \begin{itemize}
    \item ともに弧長パラメータづけ可能な$C^3$級曲線
    \item それぞれの弧長パラメータ$s\in s(I),s'\in s'(I')$について、常に$\frac{d^2\gamma(t(s))}{ds^2}\neq0$、$\frac{d^2\gamma'(t'(s'))}{ds'^2}\neq0$
    \item $\kappa(s)$、$\tau(s)$は$(\gamma,I)$の曲率と捩率
    \item $\kappa'(s')$、$\tau'(s')$は$(\gamma',I')$の曲率と捩率
    \item ある開区間$J\subset s(I)$と開区間$J'\subset s'(I')$、および、ある$s_0\in\mathbb{R}と\sigma\in\{\pm1\}$が存在し、\\
    写像 $J\to J';s\mapsto \sigma s+s_0$は全単射
  \end{itemize}
  このとき、もしある定数 $\varepsilon\in\{±1\}$ が存在し、全ての $s\in J$ に対して
  \[
    \kappa'(\sigma s+s_0) = \kappa(s) \quad \text{かつ} \quad \tau'(\sigma s+s_0) = \varepsilon \tau(s)
  \]
  が成り立つならば、曲線$(\gamma,I)$と$(\gamma',I')$は互いに合同である。
  すなわち、ある3次直交行列 $M$ とベクトル $p\in\mathbb{R}^3$ が一意に存在して、
  \[
    \det(M) = \varepsilon
  \]
  かつ、任意の$s\in J$で
  \[
    \gamma'(t'(\sigma s+s_0)) = M(\gamma(t(s))) + p
  \]
  が成り立つ。
\end{theorem}
\begin{proof}
  $\gamma$および$\gamma'$のフレネ標構をそれぞれ
  \[
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix},\quad
    \begin{pmatrix}
      e'_1(s)\\
      e'_2(s)\\
      e'_3(s)
    \end{pmatrix}
  \]
  とする。
  曲線の合同変換、およびパラメータの平行移動によって
  \begin{itemize}
    \item $s_0\in s(I)\cap s'(I')$
    \item $\gamma(t(s_0))=\gamma'(t'(s_0))$
    \item $e_i(s_0)=e'_i(s_0)$ ($i=1,2,3$)
  \end{itemize}
  であるとしてよい。
  このとき、$\gamma(t(s))=\gamma'(t'(s))$ (${}^\forall s\in s(I)\cap s'(I')$)を証明するには、
  \[
    \frac{d}{ds}(\gamma(t(s))-\gamma'(t'(s)))=0 \text{ すなわち } e_1(s)-e'_1(s)=0
  \]
  を示せばよい($\gamma(t(s_0))=\gamma'(t'(s_0))$ゆえ)。
  そのためには、$i=1,2,3$に対して
  \begin{align*}
    e_i(s)&=(\xi_{i1}(s),\xi_{i2}(s),\xi_{i3}(s))\\
    e'_i(s)&=(\xi'_{i1}(s),\xi'_{i2}(s),\xi'_{i3}(s))\\
  \end{align*}
  とおいたとき、
  \[
    \xi_{1j}(s)-\xi'_{1j}(s)=0
  \]
  を示せばよい。
  これを示すためには、$j=1,2,3$に対して、
  \[
    (\xi_{1j}(s)-\xi'_{1j}(s))^2+(\xi_{2j}(s)-\xi'_{2j}(s))^2+(\xi_{3j}(s)-\xi'_{3j}(s))^2=0
  \]
  を示せばよい。まず展開してみると、
  \begin{align*}
    &(\xi_{1j}(s)-\xi'_{1j}(s))^2+(\xi_{2j}(s)-\xi'_{2j}(s))^2+(\xi_{3j}(s)-\xi'_{3j}(s))^2\\
    =&\xi_{1j}(s)^2+\xi_{2j}(s)^2+\xi_{3j}(s)^2+\xi'_{1j}(s)^2+\xi'_{2j}(s)^2+\xi'_{3j}(s)^2\\
    &-2(\xi_{1j}(s)\xi'_{1j}(s)+\xi_{2j}(s)\xi'_{2j}(s)+\xi_{3j}(s)\xi'_{3j}(s))
  \end{align*}
  となる。
  ここでフレネ標構が正規直交基底であることを思い出せば、結局
  \begin{align*}
    &(\xi_{1j}(s)-\xi'_{1j}(s))^2+(\xi_{2j}(s)-\xi'_{2j}(s))^2+(\xi_{3j}(s)-\xi'_{3j}(s))^2\\
    =&2-2(\xi_{1j}(s)\xi'_{1j}(s)+\xi_{2j}(s)\xi'_{2j}(s)+\xi_{3j}(s)\xi'_{3j}(s))
  \end{align*}
  が成り立つ。
  ところが$s=s_0$では$e_i(s_0)=e'_i(s_0)$であるとしていたので、再びフレネ標構が正規直交基底であったことから
  \[
    \xi_{1j}(s_0)\xi'_{1j}(s_0)+\xi_{2j}(s_0)\xi'_{2j}(s_0)+\xi_{3j}(s_0)\xi'_{3j}(s_0)=1
  \]
  となり、
  \[
    (\xi_{1j}(s_0)-\xi'_{1j}(s_0))^2+(\xi_{2j}(s_0)-\xi'_{2j}(s_0))^2+(\xi_{3j}(s_0)-\xi'_{3j}(s_0))^2=0
  \]
  が従う。
  ゆえに題意を示すには、
  \[
    \frac{d}{ds}(\xi_{1j}(s)\xi'_{1j}(s)+\xi_{2j}(s)\xi'_{2j}(s)+\xi_{3j}(s)\xi'_{3j}(s))=0
  \]
  を示せばよい。
  ところがこれは、フレネ・セレの公式より、
  \begin{align*}
    \frac{d(\xi_{1j}(s)\xi'_{1j}(s))}{ds}&=\frac{d\xi_{1j}(s)}{ds}\xi'_{1j}(s)+\xi_{1j}(s)\frac{d\xi'_{1j}(s)}{ds}=\kappa(s)\xi_{2j}(s)\xi'_{1j}(s)+\kappa(s)\xi_{1j}(s)\xi'_{2j}(s)\\
    \frac{d(\xi_{2j}(s)\xi'_{2j}(s))}{ds}&=\frac{d\xi_{2j}(s)}{ds}\xi'_{2j}(s)+\xi_{2j}(s)\frac{d\xi'_{2j}(s)}{ds}\\
    &=(-\kappa(s)\xi_{1j}(s)+\tau(s)\xi_{3j}(s))\xi'_{2j}(s)+\xi_{2j}(s)(-\kappa(s)\xi'_{1j}(s)+\tau(s)\xi'_{3j}(s))\\
    &=-\kappa(s)\xi_{1j}(s)\xi'_{2j}(s)-\kappa(s)\xi_{2j}(s)\xi'_{1j}(s)+\tau(s)\xi_{2j}(s)\xi'_{3j}(s)+\tau(s)\xi_{3j}(s)\xi'_{2j}(s)\\
    \frac{d(\xi_{3j}(s)\xi'_{3j}(s))}{ds}&=\frac{d\xi_{3j}(s)}{ds}\xi'_{3j}(s)+\xi_{3j}(s)\frac{d\xi'_{3j}(s)}{ds}=-\tau(s)\xi_{2j}(s)\xi'_{3j}(s)-\tau(s)\xi_{3j}(s)\xi'_{2j}(s)
  \end{align*}
  となって従う。
\end{proof}

曲線の基本定理の厳密な主張は上記の通りとなります。
主張を正確に述べると、少々理解しづらいので、誤解を恐れず簡単にまとめましょう。
\begin{theorem}[曲線の基本定理]
  二つの曲線が合同であるための必要十分条件は、それらの曲率と捩率がそれぞれ等しいことである。
\end{theorem}



\section{曲線の基本定理を使う}

曲線の基本定理は強力な定理です。
力学などで求めた質点の軌跡に対して、曲率と捩率を求めれば、それがどんな曲線と一致するかがわかります。
例えば、先に良く知っている曲率・捩率のリストを作っておけば便利でしょう。
既に円や常螺旋、放物線や$(t,t^2,t^3)$が描く曲線のそれらは計算しています。
それ以外にも、いくつか曲率と捩率が簡単な場合にどうなるかを調べてみましょう。

\subsection{曲率ゼロの曲線は直線}

まずは曲率ゼロの曲線について。
曲率ゼロということは、全然曲がってないということで、これはもう直線しか出てこないでしょう。
これは$e_2(s)$が定義できなくなってしまうのでフレネ・セレの公式は使えないのですが、それでもしっかり証明できます。
\begin{example}
  曲線$(\gamma,I)$に関して以下の仮定を満たすとする。
  \begin{itemize}
    \item 弧長パラメータづけ可能な2階微分可能な曲線
    \item $\kappa(s)$は$(\gamma,I)$の曲率
  \end{itemize}
  さらに$\kappa(s)=0$ (${}^\forall s\in I$)を満たす場合、$(\gamma,I)$は直線である。
\end{example}
\begin{proof}
  $\kappa(s)=0$ゆえ、フレネ・セレの公式は使えないが、
  \[\frac{de_1(s)}{ds}=\kappa(s)e_2(s)=0\]
  は成り立つ。
  従って$e_1(s)=d\gamma(t(s))/ds$は定値写像である。
  $e_1(s)\neq0$であることに注意して、$v:=e_1(s)\neq0$とすれば
  \[
    \frac{d\gamma(t(s))}{ds}=v \iff \gamma(t(s))=vs+\gamma_0(\text{$\gamma_0\in\mathbb{R}^3$は積分定数})
  \]
  となり、これは直線に他ならない。
\end{proof}

\subsection{捩率ゼロの曲線はある平面に含まれる}

続いて捩率ゼロの場合です。
$\kappa(s)=0$だと上述のとおりつまらないので、フレネ・セレの公式が使える$\kappa(s)>0$の状況で考えましょう。
すると、次が成り立ちます。
\begin{example}
  曲線$(\gamma,I)$に関して以下の仮定を満たすとする。
  \begin{itemize}
    \item 弧長パラメータづけ可能な$C^3$級曲線
    \item 弧長パラメータ$s\in s(I)$について、常に$\frac{d^2\gamma(t(s))}{ds^2}\neq0$
    \item $\kappa(s)$、$\tau(s)$は$(\gamma,I)$の曲率と捩率
  \end{itemize}
  さらに$\kappa(s)>0$、$\tau(s)=0$ (${}^\forall s\in s(I)$)を満たす場合、曲線$(\gamma,I)$はある平面に含まれる。
\end{example}
\begin{proof}
  フレネ・セレの公式より、
  \[
    \frac{de_3(s)}{ds}=0
  \]
  ゆえに、$e_3(s)$は定値写像である。
  これを$n:=e_3(s)$とする。
  適当に$s_0\in s(I)$をとり、$p_0:=\gamma(t(s_0))$を通って、法線ベクトルを$n$とする平面を$H$とおく。
  すなわち、
  \[
    H:=\{p\in\mathbb{R}^3\mid(p-p_0)\cdot n=0\}
  \]
  $\gamma(t(s))$がすべて$H$に含まれていることを示すためには、
  \[
    f(s):=(\gamma(t(s))-p_0)\cdot n
  \]
  が常にゼロであることを示せばよい。
  まず仮定より
  \[
    f(s_0)=(\gamma(t(s_0))-\gamma(t(s_0)))\cdot n=0
  \]
  ゆえに、示せばよいことは
  \[
    \frac{df}{ds}=0
  \]
  である。
  これは
  \begin{align*}
    \frac{df}{ds}&=\frac{d\gamma(t(s))}{ds}\cdot n\\
    &=e_1(s)\cdot n\\
    &=e_1(s)\cdot e_3(s)\\
    &=0
  \end{align*}
  から従う。
\end{proof}
この事実は、捩率の「感覚」を向上させることに役立つでしょう。
つまり捩率とは、\textbf{曲線が平面に乗れない度合い}を測るのです。

\subsection{平面に埋め込まれる曲線の捩率はゼロ}

先ほどは「捩率ゼロならば平面に含まれる」でした。
念のためではあるのですが、「平面に含まれるならば捩率ゼロ」も示していきましょう。
\begin{example}
  曲線$(\gamma,I)$に関して以下の仮定を満たすとする。
  \begin{itemize}
    \item 弧長パラメータづけ可能な$C^3$級曲線
    \item 弧長パラメータ$s\in s(I)$について、常に$\frac{d^2\gamma(t(s))}{ds^2}\neq0$
    \item $\kappa(s)$、$\tau(s)$は$(\gamma,I)$の曲率と捩率
  \end{itemize}
  さらに、$(\gamma,I)$がある平面$H$に含まれるならば、$\tau(s)=0$。
\end{example}
\begin{proof}
  仮定より$\kappa(s)>0$ (${}^\forall s\in s(I)$)。
  いま、適当に$s_0\in s(I)$をとれば、$\gamma(I)$を含む$H$は
  \[
    H=\{p\in\mathbb{R}^3\mid (p-p_0)\cdot n=0\}
  \]
  と表わせる。ここで$n\in\mathbb{R}^3$は$H$の単位法線ベクトルである。
  仮定から、任意の$s\in s(I)$に対して
  \[
    (\gamma(t(s))-p_0)\cdot n=0
  \]
  が成り立つ。
  両辺を微分して、
  \[
    \frac{d\gamma(t(s))}{ds}\cdot n=e_1(s)\cdot n=0
  \]
  を得る。さらに微分すると、フレネ・セレの公式より
  \[
    \frac{de_1(s)}{ds}\cdot n=\kappa(s)e_2(s)\cdot n=0
  \]
  を得る。
  $\kappa(s)>0$であるから$e_2(s)\cdot n=0$であるが、これをさらに微分すれば
  \[
    \frac{de_2(s)}{ds}\cdot n=(-\kappa(s)e_1(s)+\tau(s)e_3(s))\cdot n=\tau(s)e_3(s)\cdot n=0
  \]
  を得る。ここで仮に$\tau(s)\neq0$となる$s\in s(I)$が存在したとすると、
  \[
    \begin{cases}
      e_1(s)\cdot n=0\\
      e_2(s)\cdot n=0\\
      e_3(s)\cdot n=0
    \end{cases}
  \]
  が得られてしまうが、これは行列とベクトルの積の等式
  \[
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
    n=0
  \]
  に他ならない。
  ところが
  \[
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
  \]
  は直交行列ゆえ逆行列をもつので、左からその逆行列をかければ
  \[
    n=0
  \]
  を得る。
  これは最初に$n$を$H$の単位法ベクトルとしたことに矛盾する。
  よって任意の$s\in s(I)$について$\tau(s)=0$でなければならない。
\end{proof}

これで「曲線が平面に埋め込まれる $\iff$ 捩率ゼロ」が証明できました。
ただ一言付け加えると、今回どちらもフレネ・セレの公式を使って証明しましたが、曲線の基本定理を使えば、どちらか片方を示せばもう片方も従っていました。

\subsection{曲率・捩率が定数の場合}

曲率・捩率が0である場合を見てきました。
次に簡単な場合は、それらが0ではない定数の場合です。
曲率が0の場合はつまらないので、曲率は0にならないときにどうなるか調べてみましょう。
ここまでは実は曲線の基本定理を使わず、フレネ・セレの公式だけから証明できることばかりでしたが、次の例は曲線の基本定理を使うのが楽です。

まずは捩率ゼロのときです。
\begin{example}
  曲線$(\gamma,I)$に関して以下の仮定を満たすとする。
  \begin{itemize}
    \item 弧長パラメータづけ可能な$C^3$級曲線
    \item 弧長パラメータ$s\in s(I)$について、常に$\frac{d^2\gamma(t(s))}{ds^2}\neq0$
    \item $\kappa(s)$、$\tau(s)$は$(\gamma,I)$の曲率と捩率
  \end{itemize}
  さらに、任意の$s\in s(I)$に対して
  \[
    \begin{cases}
      \kappa(s)=a>0\\
      \tau(s)=0
    \end{cases}
  \]
  を満たすとする。
  このとき、曲線$(\gamma,I)$はある円に含まれる。
\end{example}
\begin{proof}
  \[
    \gamma'(t(s))=\left(\frac1a\cos(as),\frac1a\sin(as),0\right)
  \]
  は、曲率$a$、捩率0をもつ曲線である。
  実際、これは平面$H=\{z=0\}$に含まれるため、捩率0である。
  また、曲率は
  \begin{align*}
    \kappa(s)=\left|\frac{d^2\gamma'(t(s))}{ds^2}\right|=\left|\frac{d}{ds}(-\sin(as),\cos(as),0)\right|=|(-a\cos(as),-a\sin(as),0)|=a
  \end{align*}
  となる。
  ゆえに曲線の基本定理より、曲線$(\gamma,I)$は円$(\gamma',I')$と合同であるから、題意は証明された。
\end{proof}

\begin{example}
  曲線$(\gamma,I)$に関して以下の仮定を満たすとする。
  \begin{itemize}
    \item 弧長パラメータづけ可能な$C^3$級曲線
    \item 弧長パラメータ$s\in s(I)$について、常に$\frac{d^2\gamma(t(s))}{ds^2}\neq0$
    \item $\kappa(s)$、$\tau(s)$は$(\gamma,I)$の曲率と捩率
  \end{itemize}
  さらに、任意の$s\in s(I)$に対して
  \[
    \begin{cases}
      \kappa(s)=a>0\\
      \tau(s)=b\neq0
    \end{cases}
  \]
  を満たすとする。
  このとき、曲線$(\gamma,I)$はある常螺旋に含まれる。
\end{example}
\begin{proof}
  常螺旋
  \[
    \gamma'(t)=\left(\frac{a}{a^2+b^2}\cos(t),\frac{a}{a^2+b^2}\sin(t),\frac{bt}{a^2+b^2}\right)
  \]
  を考える。
  まず、この弧長パラメータは
  \[
    s(t)=t\sqrt{\left(\frac{a}{a^2+b^2}\right)^2+\left(\frac{b}{a^2+b^2}\right)^2}=\frac{t}{\sqrt{a^2+b^2}}
  \]
  であるから、弧長パラメータ表示は
  \[
    \gamma(t(s))=\left(\frac{a}{a^2+b^2}\cos(s\sqrt{a^2+b^2}),\frac{a}{a^2+b^2}\sin(s\sqrt{a^2+b^2}),\frac{bs}{\sqrt{a^2+b^2}}\right)
  \]
  となる。この曲率を求めると
  \[
    \kappa(s)=\left|\frac{d^2\gamma(t(s))}{ds^2}\right|=|(-a\cos(s\sqrt{a^2+b^2}),-a\sin(s\sqrt{a^2+b^2}),0)|=a
  \]
  また、捩率を求めるために、まず次の計算をする
  \begin{align*}
    &\frac{d\gamma(t(s))}{ds}\times\frac{d^2\gamma(t(s))}{ds^2}\\
    =&(-\frac{a}{\sqrt{a^2+b^2}}\sin(s\sqrt{a^2+b^2}),\frac{a}{\sqrt{a^2+b^2}}\cos(s\sqrt{a^2+b^2}),\frac{b}{\sqrt{a^2+b^2}})\\
    &\times(-a\cos(s\sqrt{a^2+b^2}),-a\sin(s\sqrt{a^2+b^2}),0)\\
    =&\frac{1}{\sqrt{a^2+b^2}}(ab\sin(s\sqrt{a^2+b^2}),-ab\cos(s\sqrt{a^2+b^2}),a^2)
  \end{align*}
  また
  \[
    \frac{d^3\gamma(t(s))}{ds^3}=a\sqrt{a^2+b^2}(\sin(s\sqrt{a^2+b^2}),-\cos(s\sqrt{a^2+b^2}),0)
  \]
  ゆえに、
  \[
    \left(\frac{d\gamma(t(s))}{ds}\times\frac{d^2\gamma(t(s))}{ds^2}\right)\cdot\frac{d^3\gamma(t(s))}{ds^3}=a^2b
  \]
  となり、捩率は
  \[
    \tau(s)=\frac{\left(\frac{d\gamma(t(s))}{ds}\times\frac{d^2\gamma(t(s))}{ds^2}\right)\cdot\frac{d^3\gamma(t(s))}{ds^3}}{\left|\frac{d^2\gamma(t(s))}{ds^2}\right|^2}=\frac{a^2b}{a^2}=b
  \]
  となる。
  よって曲線の基本定理より、$(\gamma,I)$は常螺旋と合同である。
  ゆえに題意は示された。
\end{proof}




\section{曲率と捩率から曲線を作る}

フレネ・セレの公式を眺めていると、一つの興味が浮かびます。
\begin{center}
  これは微分方程式だな？じゃあ、曲率と捩率から逆に曲線がつくれるのでは？
\end{center}
実はこれは成り立ちます。
\begin{theorem}[指定した曲率・捩率をもつ曲線の存在定理]
  $I$を区間、$\kappa:I\to(0,+\infty)$を$C^1$級関数、$\tau:I\to\mathbb{R}$を連続写像とする。
  このとき、ある$C^3$級曲線$\gamma:I\to\mathbb{R}^3$が存在して、その曲率と捩率がそれぞれ$\kappa(s)$、$\tau(s)$になる。
\end{theorem}
\begin{proof}
  フレネ・セレの公式
  \[
    \frac{d}{ds}
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
    =
    \begin{pmatrix}
      0 & \kappa(s) & 0 \\
      -\kappa(s) & 0 & \tau(s) \\
      0 & -\tau(s) & 0
    \end{pmatrix}
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
  \]
  を微分方程式とみて、解の存在と一意性定理を使用する。
  この方程式は線形微分方程式ゆえ、$s_0\in I$と$e_i(s_0)$を適当に決めれば、フレネ・セレの公式を満たすフレネ標構
  \[
    \begin{pmatrix}
      e_1(s)\\
      e_2(s)\\
      e_3(s)
    \end{pmatrix}
  \]
  が一意に定まり、これらはもちろん可微分である。
  このとき、
  \[
    \gamma:I\to \mathbb{R}^3;s\mapsto\int_{s_0}^s e_1(s)ds
  \]
  とおけば、これは求める曲線である。

  実際、曲率については
  \begin{align*}
    \frac{d\gamma(s)}{ds}=\frac{d}{ds}\int_{s_0}^se_1(s)ds=e_1(s)
  \end{align*}
  ゆえに
  \begin{align*}
    \left|\frac{d^2\gamma(s)}{ds^2}\right|=\left|\frac{de_1(s)}{ds}\right|=\left|\kappa(s)e_2(s)\right|=\kappa(s)
  \end{align*}
  となり、確かめられた。
  また、捩率については
  \begin{align*}
    \frac{d\gamma(s)}{ds}&=e_1(s)\\
    \frac{d^2\gamma(s)}{ds^2}&=\frac{de_1(s)}{ds}=\kappa(s)e_2(s)\\
    \frac{d^3\gamma(s)}{ds^3}&=\frac{d(\kappa(s)e_2(s))}{ds}=\frac{d\kappa(s)}{ds}e_2(s)+\kappa(s)\frac{de_2(s)}{ds}=-\kappa(s)^2e_1(s)+\frac{d\kappa(s)}{ds}e_2(s)+\kappa(s)\tau(s)e_3(s)\\
  \end{align*}
  という計算により(このことから$\gamma$が$C^3$級であることもわかる)、捩率の$\gamma$のみによる定義式の分母にくる関数は
  \begin{align*}
    &\left(\frac{d\gamma(s)}{ds}\times\frac{d^2\gamma(s)}{ds^2}\right)\cdot\frac{d^3\gamma(s)}{ds^3}\\
    =&(e_1(s)\times\kappa(s)e_2(s))\cdot\left(-\kappa(s)^2e_1(s)+\frac{d\kappa(s)}{ds}e_2(s)+\kappa(s)\tau(s)e_3(s)\right)\\
    =&\kappa(s)e_3\cdot\left(-\kappa(s)^2e_1(s)+\frac{d\kappa(s)}{ds}e_2(s)+\kappa(s)\tau(s)e_3(s)\right)\\
    =&\kappa(s)^2\tau(s)
  \end{align*}
  となる。そして分母は
  \[
    \left|\frac{d^2\gamma(s)}{ds^2}\right|^2=\left|\kappa(s)e_2(s)\right|^2=\kappa(s)^2
  \]
  となり、従って
  \[
    \frac{\left(\frac{d\gamma(s)}{ds}\times\frac{d^2\gamma(s)}{ds^2}\right)\cdot\frac{d^3\gamma(s)}{ds^3}}{\left|\frac{d^2\gamma(s)}{ds^2}\right|^2}=\tau(s)
  \]
  が確かめられた。
\end{proof}
